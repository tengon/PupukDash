   110→function MonthlySales({ data }: { data: DashboardData }) {

   112→

   115→  const lastMonth = data.monthlySales[data.monthlySales.length - 2]
   116→  let percentageChange: number | null = null
   117→  let isIncrease = false
   118→  if (thisMonth && lastMonth && lastMonth.total > 0) {
   119→    percentageChange = Math.round(((thisMonth.total - lastMonth.total) / lastMonth.total) * 100)
   120→    isIncrease = thisMonth.total >= lastMonth.total
   121→  }
   122→

   134→                <div className="flex justify-between text-xs">

   416→                <TableRow key={f.id} className="hover:border-l-2 hover:border-l-amber-300">
   417→                  <TableCell className="text-xs">
   418→                    {i < 3 ? (
   419→                      <span className={`inline-flex h-6 w-6 items-center justify-center rounded-full text-[11px] font-bold ${medalColors[i]}`}>
   420→                        {i + 1}
   421→                      </span>
   422→                    ) : (
   423→                      <span className={`inline-flex h-6 w-6 items-center justify-center rounded-full text-[11px] font-bold bg-muted text-muted-foreground ${medalTextColors[i] || ''}`}>

   430→                      {i < 3 && (
   431→                        <span className="text-sm inline-block transition-transform duration-300 hover:scale-125 cursor-default">
   432→                          {i === 0 ? '🥇' : i === 1 ? '🥈' : '🥉'}
   433→                        </span>
   434→                      )}
   435→                      {f.name}
   436→                    </div>
   437→                  </TableCell>
   438→                  <TableCell className={`text-xs text-right font-semibold ${i < 3 ? medalTextColors[i] : ''}`}>{formatRupiah(f.totalAmount)}</TableCell>

   455→function QuickInfoCards() {
   456→  const { refreshKey } = useAppStore()
   457→
   458→  const { data: stockData } = useQuery({
   459→    queryKey: ['stock', refreshKey],
   460→    queryFn: fetchStock,
   461→  })
   462→
   463→  const { data: productsData } = useQuery({
   464→    queryKey: ['products', refreshKey],
   465→    queryFn: fetchProducts,
   466→  })
   467→
   468→  const { data: distributionsData } = useQuery({
   469→    queryKey: ['distributions', refreshKey],
   470→    queryFn: fetchDistributions,
   471→  })
   472→
   473→  const totalStock = (stockData || []).reduce((sum, s) => sum + s.quantity, 0)
   474→
   475→  const activeProducts = (productsData || []).filter(p => p.isActive)
   476→  const avgSubsidyPrice = activeProducts.length > 0
   477→    ? activeProducts.reduce((sum, p) => sum + p.subsidyPrice, 0) / activeProducts.length
   478→    : 0
   479→
   480→  const now = new Date()
   481→  const startOfMonth = new Date(now.getFullYear(), now.getMonth(), 1)
   482→  const endOfMonth = new Date(now.getFullYear(), now.getMonth() + 1, 0, 23, 59, 59)
   483→  const distThisMonth = (distributionsData || []).filter(d => {
   484→    const dDate = new Date(d.createdAt)
   485→    return dDate >= startOfMonth && dDate <= endOfMonth
   486→  }).length
   487→
   488→  const cards = [
   489→    {
   490→      title: 'Total Stok Tersedia',
   491→      value: `${formatNumber(totalStock)} kg`,
   492→      icon: Package,
   493→      color: 'text-teal-600 dark:text-teal-400',
   494→      bgColor: 'bg-teal-50 dark:bg-teal-900/20',
   495→      borderColor: 'border-teal-200 dark:border-teal-800',
   496→    },
   497→    {
   498→      title: 'Rata-rata Harga Subsidi/kg',
   499→      value: formatRupiah(Math.round(avgSubsidyPrice)),
   500→      icon: TrendingDown,
   501→      color: 'text-green-600 dark:text-green-400',
   502→      bgColor: 'bg-green-50 dark:bg-green-900/20',
   503→      borderColor: 'border-green-200 dark:border-green-800',
   504→    },
   505→    {
   506→      title: 'Distribusi Bulan Ini',
   507→      value: `${formatNumber(distThisMonth)} pengiriman`,
   508→      icon: Truck,
   509→      color: 'text-emerald-600 dark:text-emerald-400',
   510→      bgColor: 'bg-emerald-50 dark:bg-emerald-900/20',
   511→      borderColor: 'border-emerald-200 dark:border-emerald-800',
   512→    },
   513→  ]
   514→
   515→  return (
   516→    <motion.div variants={itemVariants}>
   517→      <div className="grid grid-cols-1 md:grid-cols-3 gap-3">
   518→        {cards.map((card) => (
   519→          <Card key={card.title} className={`border ${card.borderColor} hover:shadow-md transition-all duration-300 hover:-translate-y-0.5`}>
   520→            <CardContent className="p-3 flex items-center gap-3">
   521→              <div className={`flex h-10 w-10 shrink-0 items-center justify-center rounded-xl ${card.bgColor} ring-1 ring-black/5 dark:ring-white/5`}>
   522→                <card.icon className={`h-5 w-5 ${card.color}`} />
   523→              </div>
   524→              <div className="min-w-0 flex-1">
   525→                <p className="text-[10px] text-muted-foreground leading-tight truncate uppercase tracking-wide">{card.title}</p>
   526→                <p className="text-sm font-bold mt-0.5 truncate">{card.value}</p>
   527→              </div>
   528→            </CardContent>
   529→          </Card>
   530→        ))}
   531→      </div>
   532→    </motion.div>
   533→  )
