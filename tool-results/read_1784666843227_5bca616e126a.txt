     1→'use client'
     2→
     3→import { useState } from 'react'
     4→import { useQuery, useMutation, useQueryClient } from '@tanstack/react-query'
     5→import { useAppStore } from '@/lib/store'
     6→import { fetchFarmers, createFarmer, updateFarmer, deleteFarmer, fetchOrders, fetchFarmerOrders, type Farmer, type FarmerOrdersResponse } from '@/lib/api'
     7→import { formatNumber, formatDate, formatRupiah, getStatusColor, getStatusLabel } from '@/lib/format'
     8→import { Card, CardContent, CardHeader, CardTitle } from '@/components/ui/card'
     9→import { Button } from '@/components/ui/button'
    10→import { Input } from '@/components/ui/input'
    11→import { Label } from '@/components/ui/label'
    12→import { Badge } from '@/components/ui/badge'
    13→import { Skeleton } from '@/components/ui/skeleton'
    14→import {
    15→  Dialog, DialogContent, DialogDescription, DialogFooter, DialogHeader, DialogTitle,
    16→} from '@/components/ui/dialog'
    17→import {
    18→  AlertDialog, AlertDialogAction, AlertDialogCancel, AlertDialogContent, AlertDialogDescription, AlertDialogFooter, AlertDialogHeader, AlertDialogTitle,
    19→} from '@/components/ui/alert-dialog'
    20→import {
    21→  Table, TableBody, TableCell, TableHead, TableHeader, TableRow,
    22→} from '@/components/ui/table'
    23→import {
    24→  Pagination, PaginationContent, PaginationItem, PaginationLink, PaginationNext, PaginationPrevious, PaginationEllipsis,
    25→} from '@/components/ui/pagination'
    26→import { Tabs, TabsContent, TabsList, TabsTrigger } from '@/components/ui/tabs'
    27→import { ScrollArea } from '@/components/ui/scroll-area'
    28→import { Separator } from '@/components/ui/separator'
    29→import { motion } from 'framer-motion'
    30→import { Plus, Search, Pencil, Trash2, Users, Eye, Download, UserCheck, Wheat, RotateCcw, MapPin, ShoppingCart, Banknote, Package, Phone, MapPinned } from 'lucide-react'
    31→import { exportToCSV } from '@/lib/export'
    32→import { useToast } from '@/hooks/use-toast'
    33→
    34→const ITEMS_PER_PAGE = 10
    35→
    36→const emptyForm = {
    37→  nik: '', name: '', phone: '', address: '', village: '',
    38→  district: '', regency: '', province: '', landAreaHa: 0, farmerGroup: '', isActive: true,
    39→}
    40→
    41→export function FarmersView() {
    42→  const { refreshKey, triggerRefresh, setActiveTab, setPrefillFarmerId } = useAppStore()
    43→  const { toast } = useToast()
    44→  const queryClient = useQueryClient()
    45→  const [search, setSearch] = useState('')
    46→  const [dialogOpen, setDialogOpen] = useState(false)
    47→  const [deleteOpen, setDeleteOpen] = useState(false)
    48→  const [detailOpen, setDetailOpen] = useState(false)
    49→  const [editingId, setEditingId] = useState<string | null>(null)
    50→  const [deletingId, setDeletingId] = useState<string | null>(null)
    51→  const [detailFarmer, setDetailFarmer] = useState<Farmer | null>(null)
    52→  const [form, setForm] = useState(emptyForm)
    53→  const [nikError, setNikError] = useState('')
    54→  const [page, setPage] = useState(1)
    55→
    56→  const { data: farmers, isLoading } = useQuery({
    57→    queryKey: ['farmers', refreshKey],
    58→    queryFn: fetchFarmers,
    59→  })
    60→
    61→  const { data: ordersData } = useQuery({
    62→    queryKey: ['orders', refreshKey],
    63→    queryFn: fetchOrders,
    64→    enabled: detailOpen && !!detailFarmer,
    65→  })
    66→
    67→  // Purchase history dialog
    68→  const [purchaseOpen, setPurchaseOpen] = useState(false)
    69→  const [purchaseFarmerId, setPurchaseFarmerId] = useState<string | null>(null)
    70→
    71→  const { data: purchaseData, isLoading: purchaseLoading } = useQuery<FarmerOrdersResponse>({
    72→    queryKey: ['farmer-orders', purchaseFarmerId],
    73→    queryFn: () => fetchFarmerOrders(purchaseFarmerId!),
    74→    enabled: purchaseOpen && !!purchaseFarmerId,
    75→  })
    76→
    77→  const handleRowClick = (farmer: Farmer) => {
    78→    setPurchaseFarmerId(farmer.id)
    79→    setPurchaseOpen(true)
    80→  }
    81→
    82→  const orders = ordersData?.orders ?? []
    83→
    84→  const createMutation = useMutation({
    85→    mutationFn: createFarmer,
    86→    onSuccess: () => {
    87→      queryClient.invalidateQueries({ queryKey: ['farmers'] })
    88→      triggerRefresh(); setDialogOpen(false); setForm(emptyForm); setEditingId(null)
    89→      toast({ title: 'Berhasil', description: 'Petani berhasil ditambahkan' })
    90→    },
    91→    onError: (err: Error) => toast({ title: 'Gagal', description: err.message, variant: 'destructive' }),
    92→  })
    93→
    94→  const updateMutation = useMutation({
    95→    mutationFn: ({ id, data }: { id: string; data: Partial<Farmer> }) => updateFarmer(id, data),
    96→    onSuccess: () => {
    97→      queryClient.invalidateQueries({ queryKey: ['farmers'] })
    98→      triggerRefresh(); setDialogOpen(false); setForm(emptyForm); setEditingId(null)
    99→      toast({ title: 'Berhasil', description: 'Data petani berhasil diperbarui' })
   100→    },
   101→    onError: (err: Error) => toast({ title: 'Gagal', description: err.message, variant: 'destructive' }),
   102→  })
   103→
   104→  const deleteMutation = useMutation({
   105→    mutationFn: deleteFarmer,
   106→    onSuccess: () => {
   107→      queryClient.invalidateQueries({ queryKey: ['farmers'] })
   108→      triggerRefresh(); setDeleteOpen(false); setDeletingId(null)
   109→      toast({ title: 'Berhasil', description: 'Petani berhasil dihapus' })
   110→    },
   111→    onError: (err: Error) => toast({ title: 'Gagal', description: err.message, variant: 'destructive' }),
   112→  })
   113→
   114→  const validateNik = (nik: string) => {
   115→    if (!/^\d{16}$/.test(nik)) { setNikError('NIK harus 16 digit angka'); return false }
   116→    setNikError(''); return true
   117→  }
   118→
   119→  const handleOpenAdd = () => { setForm(emptyForm); setEditingId(null); setNikError(''); setDialogOpen(true) }
   120→
   121→  const handleOpenEdit = (farmer: Farmer) => {
   122→    setForm({
   123→      nik: farmer.nik, name: farmer.name, phone: farmer.phone || '', address: farmer.address || '',
   124→      village: farmer.village || '', district: farmer.district || '', regency: farmer.regency || '',
   125→      province: farmer.province || '', landAreaHa: farmer.landAreaHa || 0,
   126→      farmerGroup: farmer.farmerGroup || '', isActive: farmer.isActive,
   127→    })
   128→    setEditingId(farmer.id); setNikError(''); setDialogOpen(true)
   129→  }
   130→
   131→  const handleOpenDetail = (farmer: Farmer) => { setDetailFarmer(farmer); setDetailOpen(true) }
   132→
   133→  const handleRepeatOrder = (farmer: Farmer) => {
   134→    setPrefillFarmerId(farmer.id)
   135→    setDetailOpen(false)
   136→    setActiveTab('orders')
   137→  }
   138→
   139→  const handleSave = () => {
   140→    if (!form.name.trim()) { toast({ title: 'Validasi', description: 'Nama wajib diisi', variant: 'destructive' }); return }
   141→    if (!validateNik(form.nik)) { toast({ title: 'Validasi', description: 'NIK harus 16 digit angka', variant: 'destructive' }); return }
   142→    if (editingId) { updateMutation.mutate({ id: editingId, data: form }) }
   143→    else { createMutation.mutate(form as Parameters<typeof createFarmer>[0]) }
   144→  }
   145→
   146→  const handleDelete = () => { if (deletingId) deleteMutation.mutate(deletingId) }
   147→
   148→  const filtered = (farmers || []).filter(
   149→    (f) => f.name.toLowerCase().includes(search.toLowerCase()) || f.nik.includes(search) ||
   150→      (f.farmerGroup || '').toLowerCase().includes(search.toLowerCase()) ||
   151→      (f.regency || '').toLowerCase().includes(search.toLowerCase())
   152→  )
   153→
   154→  const totalPages = Math.max(1, Math.ceil(filtered.length / ITEMS_PER_PAGE))
   155→  const safePage = Math.min(page, totalPages)
   156→  const paged = filtered.slice((safePage - 1) * ITEMS_PER_PAGE, safePage * ITEMS_PER_PAGE)
   157→  const startIndex = (safePage - 1) * ITEMS_PER_PAGE + 1
   158→  const endIndex = Math.min(safePage * ITEMS_PER_PAGE, filtered.length)
   159→
   160→  const farmerOrders = orders.filter((o) => o.farmerId === detailFarmer?.id)
   161→
   162→  const purchaseSummary = farmerOrders.reduce(
   163→    (acc, o) => ({
   164→      totalAmount: acc.totalAmount + o.totalAmount,
   165→      totalSubsidy: acc.totalSubsidy + o.totalSubsidy,
   166→      totalKg: acc.totalKg + o.items.reduce((s, i) => s + i.quantity, 0),
   167→    }), { totalAmount: 0, totalSubsidy: 0, totalKg: 0 }
   168→  )
   169→
   170→  const activeFarmers = (farmers || []).filter((f) => f.isActive).length
   171→  const allFarmers = farmers || []
   172→  const avgLandArea = allFarmers.length > 0 ? allFarmers.reduce((sum, f) => sum + (f.landAreaHa || 0), 0) / allFarmers.length : 0
   173→
   174→  return (
   175→    <motion.div initial={{ opacity: 0, y: 8 }} animate={{ opacity: 1, y: 0 }} transition={{ duration: 0.3 }} className="space-y-4">
   176→      {!isLoading && farmers && farmers.length > 0 && (
   177→        <div className="grid grid-cols-1 sm:grid-cols-2 gap-4">
   178→          <Card className="border-l-2 border-l-emerald-500"><CardContent className="p-4 flex items-center gap-4">
   179→            <div className="flex h-10 w-10 items-center justify-center rounded-lg bg-emerald-100 dark:bg-emerald-900/40"><UserCheck className="h-5 w-5 text-emerald-600 dark:text-emerald-400" /></div>
   180→            <div><p className="text-2xl font-bold text-foreground">{formatNumber(activeFarmers)}</p><p className="text-xs text-muted-foreground">Total Petani Aktif</p></div>
   181→          </CardContent></Card>
   182→          <Card className="border-l-2 border-l-teal-500"><CardContent className="p-4 flex items-center gap-4">
   183→            <div className="flex h-10 w-10 items-center justify-center rounded-lg bg-teal-100 dark:bg-teal-900/40"><Wheat className="h-5 w-5 text-teal-600 dark:text-teal-400" /></div>
   184→            <div><p className="text-2xl font-bold text-foreground">{avgLandArea.toFixed(2)}</p><p className="text-xs text-muted-foreground">Rata-rata Luas Lahan (Ha)</p></div>
   185→          </CardContent></Card>
   186→        </div>
   187→      )}
   188→
   189→      <Card className="border-l-2 border-l-emerald-500">
   190→        <CardHeader className="pb-3">
   191→          <div className="flex flex-col sm:flex-row sm:items-center sm:justify-between gap-3">
   192→            <CardTitle className="text-base font-semibold flex items-center gap-2"><Users className="h-5 w-5" /> Data Petani</CardTitle>
   193→            <div className="flex items-center gap-2">
   194→              <div className="relative"><Search className="absolute left-2.5 top-2.5 h-4 w-4 text-muted-foreground" /><Input placeholder="Cari petani..." value={search} onChange={(e) => { setSearch(e.target.value); setPage(1) }} className="pl-9 h-9 w-full sm:w-60" /></div>
   195→              <Button variant="outline" size="sm" className="shrink-0 gap-1.5" disabled={!farmers || farmers.length === 0}
   196→                onClick={() => { exportToCSV('data-petani', ['NIK', 'Nama', 'Telepon', 'Alamat', 'Desa', 'Kecamatan', 'Kabupaten', 'Provinsi', 'Luas Lahan (Ha)', 'Kelompok Tani', 'Status'],
   197→                  (farmers || []).map((f) => [f.nik, f.name, f.phone || '', f.address || '', f.village || '', f.district || '', f.regency || '', f.province || '', f.landAreaHa || '', f.farmerGroup || '', f.isActive ? 'Aktif' : 'Tidak Aktif']),) }}>
   198→                <Download className="h-4 w-4" /><span className="hidden sm:inline">Export CSV</span>
   199→              </Button>
   200→              <Button onClick={handleOpenAdd} size="sm" className="shrink-0"><Plus className="h-4 w-4 mr-1" /> Tambah Petani</Button>
   201→            </div>
   202→          </div>
   203→        </CardHeader>
   204→        <CardContent className="p-0">
   205→          {isLoading ? (<div className="p-4 space-y-3">{Array.from({ length: 5 }).map((_, i) => (<Skeleton key={i} className="h-12 w-full" />))}</div>) : (<>
   206→            <div className="overflow-x-auto">
   207→              <Table><TableHeader><TableRow>
   208→                <TableHead className="text-xs">NIK</TableHead><TableHead className="text-xs">Nama</TableHead>
   209→                <TableHead className="text-xs hidden md:table-cell">Telepon</TableHead><TableHead className="text-xs hidden lg:table-cell">Kelompok Tani</TableHead>
   210→                <TableHead className="text-xs hidden xl:table-cell">Lokasi</TableHead><TableHead className="text-xs text-right hidden sm:table-cell">Luas Lahan</TableHead>
   211→                <TableHead className="text-xs text-right">Pesanan</TableHead><TableHead className="text-xs text-right">Aksi</TableHead>
   212→              </TableRow></TableHeader>
   213→              <TableBody>
   214→                {filtered.length === 0 ? (<TableRow><TableCell colSpan={8} className="text-center py-12"><div className="flex flex-col items-center gap-2 text-muted-foreground"><Users className="h-10 w-10 opacity-30" /><p className="text-sm font-medium">{search ? 'Tidak ada petani yang cocok' : 'Belum ada data petani'}</p></div></TableCell></TableRow>) : paged.map((farmer) => (
   215→                  <TableRow key={farmer.id} className="cursor-pointer" onClick={() => handleRowClick(farmer)}>
   216→                    <TableCell className="text-xs font-mono">{farmer.nik}</TableCell>
   217→                    <TableCell className="text-sm font-medium">{farmer.name}</TableCell>
   218→                    <TableCell className="text-xs hidden md:table-cell">{farmer.phone || '-'}</TableCell>
   219→                    <TableCell className="text-xs hidden lg:table-cell">{farmer.farmerGroup || '-'}</TableCell>
   220→                    <TableCell className="text-xs hidden xl:table-cell">
   221→                      {farmer.village || farmer.regency ? (
   222→                        <div className="flex items-center gap-1 text-muted-foreground">
   223→                          <MapPin className="h-3 w-3 shrink-0" />
   224→                          <span className="truncate max-w-[120px]">{farmer.village ? `${farmer.village}, ${farmer.district || ''}` : farmer.regency || '-'}</span>
   225→                        </div>
   226→                      ) : '-'}
   227→                    </TableCell>
   228→                    <TableCell className="text-xs text-right hidden sm:table-cell">
   229→                      <div className="flex items-center justify-end gap-1.5">
   230→                        {farmer.landAreaHa ? <span className="font-mono">{formatNumber(farmer.landAreaHa)}</span> : <span>-</span>}
   231→                        {farmer.landAreaHa && (
   232→                          <Badge variant="outline" className={`text-[9px] px-1 py-0 font-semibold ${farmer.landAreaHa < 1 ? 'bg-blue-50 text-blue-600 border-blue-200 dark:bg-blue-900/20 dark:text-blue-400 dark:border-blue-700' : farmer.landAreaHa <= 2 ? 'bg-amber-50 text-amber-600 border-amber-200 dark:bg-amber-900/20 dark:text-amber-400 dark:border-amber-700' : 'bg-green-50 text-green-600 border-green-200 dark:bg-green-900/20 dark:text-green-400 dark:border-green-700'}`}>
   233→                            {farmer.landAreaHa < 1 ? 'Kecil' : farmer.landAreaHa <= 2 ? 'Sedang' : 'Besar'}
   234→                          </Badge>
   235→                        )}
   236→                      </div>
   237→                    </TableCell>
   238→                    <TableCell className="text-xs text-right">{farmer._count?.orders || 0}</TableCell>
   239→                    <TableCell className="text-right"><div className="flex items-center justify-end gap-1" onClick={(e) => e.stopPropagation()}>
   240→                      <Button variant="ghost" size="icon" className="h-8 w-8" onClick={() => handleOpenDetail(farmer)}><Eye className="h-3.5 w-3.5" /></Button>
   241→                      <Button variant="ghost" size="icon" className="h-8 w-8" onClick={() => handleOpenEdit(farmer)}><Pencil className="h-3.5 w-3.5" /></Button>
   242→                      <Button variant="ghost" size="icon" className="h-8 w-8 text-destructive hover:text-destructive" onClick={() => { setDeletingId(farmer.id); setDeleteOpen(true) }}><Trash2 className="h-3.5 w-3.5" /></Button>
   243→                    </div></TableCell>
   244→                  </TableRow>
   245→                ))}
   246→              </TableBody></Table>
   247→            </div>
   248→            {filtered.length > ITEMS_PER_PAGE && (
   249→              <div className="flex flex-col sm:flex-row items-center justify-between gap-3 px-4 py-3 border-t">
   250→                <p className="text-xs text-muted-foreground">Menampilkan {startIndex}-{endIndex} dari {filtered.length} data</p>
   251→                <Pagination><PaginationContent>
   252→                  <PaginationItem><PaginationPrevious onClick={() => setPage((p) => Math.max(1, p - 1))} className={safePage <= 1 ? 'pointer-events-none opacity-50' : 'cursor-pointer'} /></PaginationItem>
   253→                  {safePage > 3 && (<><PaginationItem><PaginationLink onClick={() => setPage(1)} className="cursor-pointer">1</PaginationLink></PaginationItem><PaginationItem><PaginationEllipsis /></PaginationItem></>)}
   254→                  {Array.from({ length: totalPages }, (_, i) => i + 1).filter((p) => p >= safePage - 1 && p <= safePage + 1).map((p) => (<PaginationItem key={p}><PaginationLink isActive={p === safePage} onClick={() => setPage(p)} className="cursor-pointer">{p}</PaginationLink></PaginationItem>))}
   255→                  {safePage < totalPages - 2 && (<><PaginationItem><PaginationEllipsis /></PaginationItem><PaginationItem><PaginationLink onClick={() => setPage(totalPages)} className="cursor-pointer">{totalPages}</PaginationLink></PaginationItem></>)}
   256→                  <PaginationItem><PaginationNext onClick={() => setPage((p) => Math.min(totalPages, p + 1))} className={safePage >= totalPages ? 'pointer-events-none opacity-50' : 'cursor-pointer'} /></PaginationItem>
   257→                </PaginationContent></Pagination>
   258→              </div>
   259→            )}
   260→          </>)}
   261→        </CardContent>
   262→      </Card>
   263→
   264→      {/* Add/Edit Dialog */}
   265→      <Dialog open={dialogOpen} onOpenChange={setDialogOpen}><DialogContent className="sm:max-w-lg max-h-[90vh] overflow-y-auto"><DialogHeader><DialogTitle>{editingId ? 'Edit Data Petani' : 'Tambah Petani Baru'}</DialogTitle><DialogDescription>{editingId ? 'Perbarui informasi petani' : 'Isi data petani baru'}</DialogDescription></DialogHeader>
   266→        <div className="grid gap-3 py-2">
   267→          <div className="grid grid-cols-2 gap-3"><div className="grid gap-2"><Label htmlFor="nik">NIK *</Label><Input id="nik" value={form.nik} onChange={(e) => { setForm({ ...form, nik: e.target.value.replace(/\D/g, '').slice(0, 16) }); if (nikError) validateNik(e.target.value.replace(/\D/g, '').slice(0, 16)) }} placeholder="16 digit NIK" maxLength={16} />{nikError && <p className="text-xs text-destructive">{nikError}</p>}</div><div className="grid gap-2"><Label htmlFor="fname">Nama Lengkap *</Label><Input id="fname" value={form.name} onChange={(e) => setForm({ ...form, name: e.target.value })} placeholder="Nama lengkap" /></div></div>
   268→          <div className="grid grid-cols-2 gap-3"><div className="grid gap-2"><Label htmlFor="phone">No. HP</Label><Input id="phone" value={form.phone} onChange={(e) => setForm({ ...form, phone: e.target.value })} placeholder="08xxxxxxxxxx" /></div><div className="grid gap-2"><Label htmlFor="landArea">Luas Lahan (Ha)</Label><Input id="landArea" type="number" value={form.landAreaHa || ''} onChange={(e) => setForm({ ...form, landAreaHa: parseFloat(e.target.value) || 0 })} /></div></div>
   269→          <div className="grid grid-cols-2 gap-3"><div className="grid gap-2"><Label htmlFor="farmerGroup">Kelompok Tani</Label><Input id="farmerGroup" value={form.farmerGroup} onChange={(e) => setForm({ ...form, farmerGroup: e.target.value })} placeholder="Nama kelompok tani" /></div><div className="grid gap-2"><Label htmlFor="village">Desa</Label><Input id="village" value={form.village} onChange={(e) => setForm({ ...form, village: e.target.value })} /></div></div>
   270→          <div className="grid grid-cols-2 gap-3"><div className="grid gap-2"><Label htmlFor="district">Kecamatan</Label><Input id="district" value={form.district} onChange={(e) => setForm({ ...form, district: e.target.value })} /></div><div className="grid gap-2"><Label htmlFor="regency">Kabupaten</Label><Input id="regency" value={form.regency} onChange={(e) => setForm({ ...form, regency: e.target.value })} /></div></div>
   271→          <div className="grid gap-2"><Label htmlFor="address">Alamat</Label><Input id="address" value={form.address} onChange={(e) => setForm({ ...form, address: e.target.value })} placeholder="Alamat lengkap" /></div>
   272→          <div className="grid gap-2"><Label htmlFor="province">Provinsi</Label><Input id="province" value={form.province} onChange={(e) => setForm({ ...form, province: e.target.value })} /></div>
   273→        </div>
   274→        <DialogFooter><Button variant="outline" onClick={() => setDialogOpen(false)}>Batal</Button><Button onClick={handleSave} disabled={createMutation.isPending || updateMutation.isPending}>{createMutation.isPending || updateMutation.isPending ? 'Menyimpan...' : 'Simpan'}</Button></DialogFooter>
   275→      </DialogContent></Dialog>
   276→
   277→      {/* Detail Dialog */}
   278→      <Dialog open={detailOpen} onOpenChange={setDetailOpen}><DialogContent className="sm:max-w-lg max-h-[90vh] overflow-y-auto"><DialogHeader><DialogTitle>Detail Petani</DialogTitle>{detailFarmer && <DialogDescription className="sr-only">Detail informasi {detailFarmer.name}</DialogDescription>}</DialogHeader>
   279→        {detailFarmer && (<div className="space-y-4">
   280→          <Tabs defaultValue="info">
   281→            <TabsList className="w-full"><TabsTrigger value="info" className="flex-1">Informasi</TabsTrigger><TabsTrigger value="orders" className="flex-1">Pesanan ({farmerOrders.length})</TabsTrigger></TabsList>
   282→            <TabsContent value="info" className="mt-3 space-y-4">
   283→              <div className="grid grid-cols-2 gap-3 text-sm">
   284→                <div><span className="text-muted-foreground">NIK:</span> <span className="font-mono">{detailFarmer.nik}</span></div>
   285→                <div><span className="text-muted-foreground">Nama:</span> <span className="font-medium">{detailFarmer.name}</span></div>
   286→                <div><span className="text-muted-foreground">No. HP:</span> {detailFarmer.phone || '-'}</div>
   287→                <div><span className="text-muted-foreground">Luas Lahan:</span> {detailFarmer.landAreaHa ? `${formatNumber(detailFarmer.landAreaHa)} Ha` : '-'}</div>
   288→                <div><span className="text-muted-foreground">Kel. Tani:</span> {detailFarmer.farmerGroup || '-'}</div>
   289→                <div><span className="text-muted-foreground">Status:</span> <Badge variant="outline" className={getStatusColor(detailFarmer.isActive ? 'ACTIVE' : 'INACTIVE')}>{detailFarmer.isActive ? 'Aktif' : 'Tidak Aktif'}</Badge></div>
