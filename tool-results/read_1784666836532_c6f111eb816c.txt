    23→function getGreeting(): { greeting: string; icon: typeof Sun } {
    24→  const hour = new Date().getHours()
    25→  if (hour < 11) return { greeting: 'Selamat Pagi', icon: Sun }
    26→  if (hour < 15) return { greeting: 'Selamat Siang', icon: CloudSun }
    27→  if (hour < 18) return { greeting: 'Selamat Sore', icon: CloudSun }
    28→  return { greeting: 'Selamat Malam', icon: Moon }

   100→function MonthlySales({ data }: { data: DashboardData }) {
   101→  const maxTotal = Math.max(...data.monthlySales.map((m) => m.total), 1)
   102→
   103→  // Comparison: this month vs last month
   104→  const thisMonth = data.monthlySales[data.monthlySales.length - 1]
   105→  const lastMonth = data.monthlySales[data.monthlySales.length - 2]
   106→  let percentageChange: number | null = null
   107→  let isIncrease = false
   108→  if (thisMonth && lastMonth && lastMonth.total > 0) {
   109→    percentageChange = Math.round(((thisMonth.total - lastMonth.total) / lastMonth.total) * 100)
   110→    isIncrease = thisMonth.total >= lastMonth.total
   111→  }
   112→
   113→  return (
   114→    <motion.div variants={itemVariants}>
   115→      <Card className="hover:shadow-lg transition-all duration-300" style={{ boxShadow: 'var(--shadow-sm)' }}>
   116→        <CardHeader className="pb-2">
   117→          <CardTitle className="text-sm font-semibold">Penjualan Bulanan</CardTitle>
   118→          <CardDescription>6 bulan terakhir</CardDescription>
   119→        </CardHeader>
   120→        <CardContent>
   121→          <div className="space-y-3">
   122→            {data.monthlySales.map((m) => (
   123→              <div key={m.month} className="space-y-1">
   124→                <div className="flex justify-between text-xs">
   125→                  <span className="text-muted-foreground">{m.month}</span>
   126→                  <span className="font-medium">{formatRupiah(m.total)}</span>
   127→                </div>
   128→                <Progress value={(m.total / maxTotal) * 100} className="h-2" />
   129→              </div>
   130→            ))}
   131→          </div>
   132→
   133→          {/* Comparison Section */}
   134→          {percentageChange !== null && (
   135→            <div className="mt-4 pt-4 border-t">
   136→              <p className="text-[10px] font-medium text-muted-foreground uppercase tracking-wide mb-2">Perbandingan Bulanan</p>
   137→              <div className="flex items-center gap-3">
   138→                <div className="flex items-center gap-2">
   139→                  {isIncrease ? (
   140→                    <div className="flex h-8 w-8 items-center justify-center rounded-full bg-green-100 dark:bg-green-900/30">
   141→                      <ArrowUp className="h-4 w-4 text-green-600 dark:text-green-400" />
   142→                    </div>
   143→                  ) : (
   144→                    <div className="flex h-8 w-8 items-center justify-center rounded-full bg-red-100 dark:bg-red-900/30">
   145→                      <ArrowDown className="h-4 w-4 text-red-600 dark:text-red-400" />
   146→                    </div>
   147→                  )}
   148→                  <div>
   149→                    <p className={`text-sm font-bold ${isIncrease ? 'text-green-600 dark:text-green-400' : 'text-red-600 dark:text-red-400'}`}>
   150→                      {isIncrease ? '\u2191' : '\u2193'} {Math.abs(percentageChange)}%
   151→                    </p>
   152→                    <p className="text-[10px] text-muted-foreground">Bulan ini vs bulan lalu</p>
   153→                  </div>
   154→                </div>
   155→                <div className="flex-1 border-l pl-3">
   156→                  <p className="text-[10px] text-muted-foreground">Bulan lalu</p>
   157→                  <p className="text-xs font-mono">{formatRupiah(lastMonth.total)}</p>
   158→                </div>
   159→                <div className="flex-1 border-l pl-3">
   160→                  <p className="text-[10px] text-muted-foreground">Bulan ini</p>
   161→                  <p className="text-xs font-mono font-semibold">{formatRupiah(thisMonth.total)}</p>

   440→function QuickInfoCards() {
   441→  const { refreshKey } = useAppStore()
   442→
   443→  const { data: stockData } = useQuery({
   444→    queryKey: ['stock', refreshKey],
   445→    queryFn: fetchStock,
   446→  })
   447→
   448→  const { data: productsData } = useQuery({
   449→    queryKey: ['products', refreshKey],
   450→    queryFn: fetchProducts,
   451→  })
   452→
   453→  const { data: distributionsData } = useQuery({
   454→    queryKey: ['distributions', refreshKey],
   455→    queryFn: fetchDistributions,
   456→  })
   457→
   458→  const totalStock = (stockData || []).reduce((sum, s) => sum + s.quantity, 0)
   459→
   460→  const activeProducts = (productsData || []).filter(p => p.isActive)
   461→  const avgSubsidyPrice = activeProducts.length > 0
   462→    ? activeProducts.reduce((sum, p) => sum + p.subsidyPrice, 0) / activeProducts.length
   463→    : 0
   464→
   465→  const now = new Date()
   466→  const startOfMonth = new Date(now.getFullYear(), now.getMonth(), 1)
   467→  const endOfMonth = new Date(now.getFullYear(), now.getMonth() + 1, 0, 23, 59, 59)
   468→  const distThisMonth = (distributionsData || []).filter(d => {
   469→    const dDate = new Date(d.createdAt)
   470→    return dDate >= startOfMonth && dDate <= endOfMonth
   471→  }).length
   472→
   473→  const cards = [
   474→    {
   475→      title: 'Total Stok Tersedia',
   476→      value: `${formatNumber(totalStock)} kg`,
   477→      icon: Package,
   478→      color: 'text-teal-600 dark:text-teal-400',
   479→      bgColor: 'bg-teal-50 dark:bg-teal-900/20',
   480→      borderColor: 'border-teal-200 dark:border-teal-800',
   481→    },
   482→    {
   483→      title: 'Rata-rata Harga Subsidi/kg',
   484→      value: formatRupiah(Math.round(avgSubsidyPrice)),
   485→      icon: TrendingDown,
   486→      color: 'text-green-600 dark:text-green-400',
   487→      bgColor: 'bg-green-50 dark:bg-green-900/20',
   488→      borderColor: 'border-green-200 dark:border-green-800',
   489→    },
   490→    {
   491→      title: 'Distribusi Bulan Ini',
   492→      value: `${formatNumber(distThisMonth)} pengiriman`,
   493→      icon: Truck,
   494→      color: 'text-emerald-600 dark:text-emerald-400',
   495→      bgColor: 'bg-emerald-50 dark:bg-emerald-900/20',
   496→      borderColor: 'border-emerald-200 dark:border-emerald-800',
   497→    },
   498→  ]
   499→
   500→  return (
   501→    <motion.div variants={itemVariants}>
   502→      <div className="grid grid-cols-1 md:grid-cols-3 gap-3">
   503→        {cards.map((card) => (
   504→          <Card key={card.title} className={`border ${card.borderColor} hover:shadow-md transition-all duration-300 hover:-translate-y-0.5`}>
   505→            <CardContent className="p-3 flex items-center gap-3">
   506→              <div className={`flex h-10 w-10 shrink-0 items-center justify-center rounded-xl ${card.bgColor} ring-1 ring-black/5 dark:ring-white/5`}>
   507→                <card.icon className={`h-5 w-5 ${card.color}`} />
   508→              </div>
   509→              <div className="min-w-0 flex-1">
   510→                <p className="text-[10px] text-muted-foreground leading-tight truncate uppercase tracking-wide">{card.title}</p>
   511→                <p className="text-sm font-bold mt-0.5 truncate">{card.value}</p>
   512→              </div>
   513→            </CardContent>
   514→          </Card>
   515→        ))}
