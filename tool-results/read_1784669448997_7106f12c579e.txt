   109→function MonthlySales({ data }: { data: DashboardData }) {
   110→  const maxTotal = Math.max(...data.monthlySales.map((m) => m.total), 1)
   111→
   112→  // Comparison: this month vs last month
   113→  const thisMonth = data.monthlySales[data.monthlySales.length - 1]
   114→  const lastMonth = data.monthlySales[data.monthlySales.length - 2]
   115→  let percentageChange: number | null = null
   116→  let isIncrease = false
   117→  if (thisMonth && lastMonth && lastMonth.total > 0) {
   118→    percentageChange = Math.round(((thisMonth.total - lastMonth.total) / lastMonth.total) * 100)
   119→    isIncrease = thisMonth.total >= lastMonth.total
   120→  }
   121→
   122→  return (
   123→    <motion.div variants={itemVariants}>
   124→      <Card className="hover:shadow-lg transition-all duration-300" style={{ boxShadow: 'var(--shadow-sm)' }}>

   130→          <div className="space-y-3">
   131→            {data.monthlySales.map((m) => (
   132→              <div key={m.month} className="space-y-1">
   133→                <div className="flex justify-between text-xs">
   134→                  <span className="text-muted-foreground">{m.month}</span>
   135→                  <span className="font-medium">{formatRupiah(m.total)}</span>
   136→                </div>
   137→                <Progress value={(m.total / maxTotal) * 100} className="h-3 rounded-full" />

   454→function QuickInfoCards() {
   455→  const { refreshKey } = useAppStore()
   456→
   457→  const { data: stockData } = useQuery({
   458→    queryKey: ['stock', refreshKey],
   459→    queryFn: fetchStock,
   460→  })
   461→
   462→  const { data: productsData } = useQuery({
   463→    queryKey: ['products', refreshKey],
   464→    queryFn: fetchProducts,
   465→  })
   466→
   467→  const { data: distributionsData } = useQuery({
   468→    queryKey: ['distributions', refreshKey],
   469→    queryFn: fetchDistributions,
   470→  })
   471→
   472→  const totalStock = (stockData || []).reduce((sum, s) => sum + s.quantity, 0)
   473→
   474→  const activeProducts = (productsData || []).filter(p => p.isActive)
   475→  const avgSubsidyPrice = activeProducts.length > 0
   476→    ? activeProducts.reduce((sum, p) => sum + p.subsidyPrice, 0) / activeProducts.length
   477→    : 0
   478→
   479→  const now = new Date()
   480→  const startOfMonth = new Date(now.getFullYear(), now.getMonth(), 1)
   481→  const endOfMonth = new Date(now.getFullYear(), now.getMonth() + 1, 0, 23, 59, 59)
   482→  const distThisMonth = (distributionsData || []).filter(d => {
   483→    const dDate = new Date(d.createdAt)
   484→    return dDate >= startOfMonth && dDate <= endOfMonth
   485→  }).length
   486→
   487→  const cards = [
   488→    {
   489→      title: 'Total Stok Tersedia',
   490→      value: `${formatNumber(totalStock)} kg`,
   491→      icon: Package,
   492→      color: 'text-teal-600 dark:text-teal-400',
   493→      bgColor: 'bg-teal-50 dark:bg-teal-900/20',
   494→      borderColor: 'border-teal-200 dark:border-teal-800',
   495→    },
   496→    {
   497→      title: 'Rata-rata Harga Subsidi/kg',
   498→      value: formatRupiah(Math.round(avgSubsidyPrice)),
   499→      icon: TrendingDown,
   500→      color: 'text-green-600 dark:text-green-400',
   501→      bgColor: 'bg-green-50 dark:bg-green-900/20',
   502→      borderColor: 'border-green-200 dark:border-green-800',
   503→    },
   504→    {
   505→      title: 'Distribusi Bulan Ini',
   506→      value: `${formatNumber(distThisMonth)} pengiriman`,
   507→      icon: Truck,
   508→      color: 'text-emerald-600 dark:text-emerald-400',
   509→      bgColor: 'bg-emerald-50 dark:bg-emerald-900/20',
   510→      borderColor: 'border-emerald-200 dark:border-emerald-800',
   511→    },
   512→  ]
   513→
   514→  return (
   515→    <motion.div variants={itemVariants}>
   516→      <div className="grid grid-cols-1 md:grid-cols-3 gap-3">
   517→        {cards.map((card) => (
   518→          <Card key={card.title} className={`border ${card.borderColor} hover:shadow-md transition-all duration-300 hover:-translate-y-0.5`}>
   519→            <CardContent className="p-3 flex items-center gap-3">
   520→              <div className={`flex h-10 w-10 shrink-0 items-center justify-center rounded-xl ${card.bgColor} ring-1 ring-black/5 dark:ring-white/5`}>
   521→                <card.icon className={`h-5 w-5 ${card.color}`} />
   522→              </div>
   523→              <div className="min-w-0 flex-1">
   524→                <p className="text-[10px] text-muted-foreground leading-tight truncate uppercase tracking-wide">{card.title}</p>
   525→                <p className="text-sm font-bold mt-0.5 truncate">{card.value}</p>
   526→              </div>
   527→            </CardContent>
   528→          </Card>
   529→        ))}
   530→      </div>
   531→    </motion.div>
   532→  )
