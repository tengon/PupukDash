   326→              </div>

   333→      <Dialog open={dialogOpen} onOpenChange={setDialogOpen}><DialogContent className="sm:max-w-lg max-h-[90vh] overflow-y-auto"><DialogHeader><DialogTitle>{editingId ? 'Edit Data Petani' : 'Tambah Petani Baru'}</DialogTitle><DialogDescription>{editingId ? 'Perbarui informasi petani' : 'Isi data petani baru'}</DialogDescription></DialogHeader>
