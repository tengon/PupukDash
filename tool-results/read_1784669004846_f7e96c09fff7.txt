   110→// ─── Daily Sales Mini Chart ──────────────────────────────────────────────────

   112→function DailySalesChart({ dailySales }: { dailySales: MonthlyReportData['dailySales'] }) {
   113→  const maxRevenue = useMemo(
   114→    () => Math.max(...dailySales.map((d) => d.revenue), 1),
   115→    [dailySales],
   116→  )
   117→
   118→  // Build an array of 31 entries (1-indexed day), fill missing with zeros
   119→  const days = useMemo(() => {
   120→    const map = new Map(dailySales.map((d) => [new Date(d.date).getDate(), d]))
   121→    return Array.from({ length: 31 }, (_, i) => {
   122→      const day = i + 1
   123→      const sale = map.get(day)
   124→      return {
   125→        day,
   126→        orders: sale?.orders ?? 0,
   127→        kg: sale?.kg ?? 0,
   128→        revenue: sale?.revenue ?? 0,
   129→      }
   130→    })
   131→  }, [dailySales])
   132→
   133→  return (
   134→    <Card>
   135→      <CardHeader className="pb-2">
   136→        <CardTitle className="text-sm font-medium text-muted-foreground">
   137→          Penjualan Harian
   138→        </CardTitle>
   139→      </CardHeader>
   140→      <CardContent className="pt-0">
   141→        <div className="flex items-end gap-[2px] h-24 print:h-16">
   142→          {days.map((d) => {
   143→            const height = d.revenue > 0 ? Math.max((d.revenue / maxRevenue) * 100, 4) : 2
   144→            return (
   145→              <div
   146→                key={d.day}
   147→                className="group relative flex-1 flex flex-col items-center justify-end h-full"
   148→              >
   149→                {/* Tooltip */}
   150→                {d.revenue > 0 && (
   151→                  <div className="absolute bottom-full mb-1 hidden group-hover:block z-10">
   152→                    <div className="bg-popover text-popover-foreground border rounded-md shadow-md px-2 py-1 text-[10px] whitespace-nowrap">
   153→                      <div className="font-medium">Tgl {d.day}</div>
   154→                      <div>{formatRupiah(d.revenue)}</div>
   155→                      <div>{formatNumber(d.kg)} kg · {d.orders} pesanan</div>
   156→                    </div>
   157→                  </div>
   158→                )}
   159→                <div
   160→                  className={`w-full rounded-t-sm transition-colors ${
   161→                    d.revenue > 0
   162→                      ? 'bg-green-500/80 hover:bg-green-500'
   163→                      : 'bg-muted-foreground/10'
   164→                  }`}
   165→                  style={{ height: `${height}%` }}
   166→                />
   167→              </div>
   168→            )
   169→          })}
   170→        </div>
   171→        <div className="flex justify-between mt-1">
   172→          <span className="text-[9px] text-muted-foreground">1</span>
   173→          <span className="text-[9px] text-muted-foreground">15</span>
   174→          <span className="text-[9px] text-muted-foreground">31</span>
   175→        </div>
   176→      </CardContent>
   177→    </Card>
   178→  )
   179→}

   257→      w.totalRevenue,
   258→      w.totalSubsidy,
   259→    ])
   260→

   263→      'No', 'Nama', 'NIK', 'Pesanan', 'Total (kg)', 'Total (Rp)', 'Subsidi (Rp)',

   281→      ['Pesanan Selesai', report.summary.completedOrders],

   283→      ['Total Berat Terjual (kg)', report.summary.totalKgSold],
   284→      ['Total Pendapatan (Rp)', report.summary.totalRevenue],
   285→      ['Total Subsidi (Rp)', report.summary.totalSubsidy],
   286→      ['Petani Dilayani', report.summary.totalFarmersServed],

   291→      ['== PER GUDANG =='],
