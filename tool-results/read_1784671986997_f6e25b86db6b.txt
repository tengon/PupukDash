    89→function OrderStatusTimeline({ status }: { status: string }) {
    90→  if (status === 'CANCELLED') {
    91→    return (
    92→      <div className="flex items-center justify-center py-4">
    93→        <Badge variant="outline" className="bg-red-100 text-red-700 border-red-200 dark:bg-red-900/30 dark:text-red-400 dark:border-red-800 px-3 py-1">
    94→          Pesanan Dibatalkan
    95→        </Badge>
    96→      </div>
    97→    )
    98→  }
    99→
   100→  const currentIdx = STATUS_STEPS.indexOf(status as typeof STATUS_STEPS[number])
   101→
   102→  return (
   103→    <div className="flex items-center justify-center py-4 px-2">
   104→      <div className="flex items-center gap-0 w-full max-w-xs">
   105→        {STATUS_STEPS.map((step, idx) => {
   106→          const StepIcon = STEP_ICONS[step]
   107→          const isCompleted = idx < currentIdx
   108→          const isCurrent = idx === currentIdx
   109→          const isFuture = idx > currentIdx
   110→
   111→          return (
   112→            <div key={step} className="flex items-center flex-1 last:flex-none">
   113→              <div className="flex flex-col items-center">
   114→                <div
   115→                  className={`flex h-8 w-8 items-center justify-center rounded-full border-2 transition-all ${
   116→                    isCompleted
   117→                      ? 'bg-green-500 border-green-500 text-white'
   118→                      : isCurrent
   119→                        ? 'bg-green-100 border-green-500 text-green-700 dark:bg-green-900/40 dark:text-green-400 shadow-[0_0_8px_rgba(34,197,94,0.4)]'
   120→                        : 'bg-muted border-muted-foreground/30 text-muted-foreground/50'
   121→                  }`}
   122→                >
   123→                  <StepIcon className="h-4 w-4" />
   124→                </div>
   125→                <span className={`text-[10px] mt-1 font-medium ${
   126→                  isCompleted || isCurrent ? 'text-green-700 dark:text-green-400' : 'text-muted-foreground'
   127→                }`}>
   128→                  {STEP_LABELS[step]}
   129→                </span>
   130→              </div>
   131→              {idx < STATUS_STEPS.length - 1 && (
   132→                <div className={`flex-1 h-0.5 mx-2 mb-4 rounded-full transition-all ${
   133→                  idx < currentIdx
   134→                    ? 'bg-green-500'
   135→                    : 'bg-muted-foreground/20'
   136→                }`} />
   137→              )}
   138→            </div>
   139→          )
   140→        })}
   141→      </div>
   142→    </div>
   143→  )
   144→}
