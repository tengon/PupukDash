    88→    88→function OrderStatusTimeline({ status }: { status: string }) {
    89→    89→  if (status === 'CANCELLED') {
    90→    90→    return (
    91→    91→      <div className="flex items-center justify-center py-4">
    92→    92→        <Badge variant="outline" className="bg-red-100 text-red-700 border-red-200 dark:bg-red-900/30 dark:text-red-400 dark:border-red-800 px-3 py-1">
    93→    93→          Pesanan Dibatalkan
    94→    94→        </Badge>
    95→    95→      </div>
    96→    96→    )
    97→    97→  }
    98→    98→
    99→    99→  const currentIdx = STATUS_STEPS.indexOf(status as typeof STATUS_STEPS[number])
   100→   100→
   101→   101→  return (
   102→   102→    <div className="flex items-center justify-center py-4 px-2">
   103→   103→      <div className="flex items-center gap-0 w-full max-w-xs">
   104→   104→        {STATUS_STEPS.map((step, idx) => {
   105→   105→          const StepIcon = STEP_ICONS[step]
   106→   106→          const isCompleted = idx < currentIdx
   107→   107→          const isCurrent = idx === currentIdx
   108→   108→          const isFuture = idx > currentIdx
   109→   109→
   110→   110→          return (
   111→   111→            <div key={step} className="flex items-center flex-1 last:flex-none">
   112→   112→              <div className="flex flex-col items-center">
   113→   113→                <div
   114→   114→                  className={`flex h-8 w-8 items-center justify-center rounded-full border-2 transition-all ${
   115→   115→                    isCompleted
   116→   116→                      ? 'bg-green-500 border-green-500 text-white'
   117→   117→                      : isCurrent
   118→   118→                        ? 'bg-green-100 border-green-500 text-green-700 dark:bg-green-900/40 dark:text-green-400 shadow-[0_0_8px_rgba(34,197,94,0.4)]'
   119→   119→                        : 'bg-muted border-muted-foreground/30 text-muted-foreground/50'
   120→   120→                  }`}
   121→   121→                >
   122→   122→                  <StepIcon className="h-4 w-4" />
   123→   123→                </div>
   124→   124→                <span className={`text-[10px] mt-1 font-medium ${
   125→   125→                  isCompleted || isCurrent ? 'text-green-700 dark:text-green-400' : 'text-muted-foreground'
   126→   126→                }`}>
   127→   127→                  {STEP_LABELS[step]}
   128→   128→                </span>
   129→   129→              </div>
   130→   130→              {idx < STATUS_STEPS.length - 1 && (
   131→   131→                <div className={`flex-1 h-0.5 mx-2 mb-4 rounded-full transition-all ${
   132→   132→                  idx < currentIdx
   133→   133→                    ? 'bg-green-500'
   134→   134→                    : 'bg-muted-foreground/20'
   135→   135→                }`} />
   136→   136→              )}
   137→   137→            </div>
   138→   138→          )
   139→   139→        })}
   140→   140→      </div>
   141→   141→    </div>
   142→   142→  )
   143→   143→}
