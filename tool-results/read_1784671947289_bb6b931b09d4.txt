    61→    61→function StockLevelBar({ quantity, minStock }: { quantity: number; minStock: number }) {
    62→    62→  const ratio = quantity / minStock
    63→    63→  const percent = Math.min((ratio / 2) * 100, 100)
    64→    64→
    65→    65→  let barColor = 'bg-green-500'
    66→    66→  let bgColor = 'bg-green-100 dark:bg-green-900/30'
    67→    67→  if (ratio < 1) {
    68→    68→    barColor = 'bg-red-500'
    69→    69→    bgColor = 'bg-red-100 dark:bg-red-900/30'
    70→    70→  } else if (ratio < 1.5) {
    71→    71→    barColor = 'bg-yellow-500'
    72→    72→    bgColor = 'bg-yellow-100 dark:bg-yellow-900/30'
    73→    73→  }
    74→    74→
    75→    75→  return (
    76→    76→    <div className={`w-full h-2 rounded-full ${bgColor} overflow-hidden`}>
    77→    77→      <div
    78→    78→        className={`h-full rounded-full ${barColor} transition-all duration-500`}
    79→    79→        style={{ width: `${percent}%` }}
    80→    80→      />
    81→    81→    </div>
    82→    82→  )
    83→    83→}

    85→    85→function FillIndicatorBar({ quantity }: { quantity: number }) {
    86→    86→  const fillPercent = Math.min((quantity / MAX_CAPACITY) * 100, 100)
    87→    87→
    88→    88→  let fillColor = 'bg-green-500'
    89→    89→  if (fillPercent < 20) fillColor = 'bg-red-500'
    90→    90→  else if (fillPercent < 40) fillColor = 'bg-yellow-500'
    91→    91→
    92→    92→  return (
    93→    93→    <div className="stock-fill-bar mt-3">
    94→    94→      <div
    95→    95→        className={`absolute bottom-0 left-0 h-full rounded-full transition-all duration-700 ${fillColor}`}
    96→    96→        style={{ width: `${fillPercent}%` }}
    97→    97→      />
    98→    98→    </div>
    99→    99→  )
   100→   100→}

   102→   102→function CapacityRing({ quantity, maxCapacity = 10000 }: { quantity: number; maxCapacity?: number }) {
   103→   103→  const radius = 28
   104→   104→  const stroke = 4
   105→   105→  const size = (radius + stroke) * 2
   106→   106→  const circumference = 2 * Math.PI * radius
   107→   107→  const percent = Math.min(quantity / maxCapacity, 1)
   108→   108→  const filled = percent * circumference
   109→   109→
   110→   110→  let ringColor = 'text-green-500'
   111→   111→  if (percent < 0.2) ringColor = 'text-red-500'
   112→   112→  else if (percent < 0.4) ringColor = 'text-yellow-500'
   113→   113→
   114→   114→  return (
   115→   115→    <div className="relative inline-flex items-center justify-center" style={{ width: size, height: size }}>
   116→   116→      <svg width={size} height={size} className="-rotate-90">
   117→   117→        <circle cx={radius + stroke} cy={radius + stroke} r={radius} fill="none" className="stroke-muted" strokeWidth={stroke} />
   118→   118→        <circle cx={radius + stroke} cy={radius + stroke} r={radius} fill="none" className={`${ringColor} transition-all duration-700`} strokeWidth={stroke} strokeLinecap="round" strokeDasharray={`${filled} ${circumference}`} />
   119→   119→      </svg>
   120→   120→      <span className="absolute text-[10px] font-bold tabular-nums">{Math.round(percent * 100)}%</span>
   121→   121→    </div>
   122→   122→  )
   123→   123→}
